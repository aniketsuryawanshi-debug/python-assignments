# 23. Electricity Bill Calculation
def q23(units):
    print(f"Q23. Units Consumed: {units}")
    if units <= 100: bill = units * 1.5
    elif units <= 300: bill = (100 * 1.5) + ((units - 100) * 2.5)
    else: bill = (100 * 1.5) + (200 * 2.5) + ((units - 300) * 4.0)
    print(f"Result: Total Bill = ${bill}\n")

# 40. Password Validity (Nested If)
def q40(pwd):
    print(f"Q40. Password: '{pwd}'")
    if len(pwd) >= 8:
        if any(c.isdigit() for c in pwd): print("Result: Valid Password\n")
        else: print("Result: Invalid (Must contain at least one digit)\n")
    else: print("Result: Invalid (Too short)\n")

# Match Case Programs (41 - 50)
def q41(day):
    print(f"Q41. Day: {day}")
    match day:
        case 1: res = "Monday"
        case 2: res = "Tuesday"
        case 3: res = "Wednesday"
        case 4: res = "Thursday"
        case 5: res = "Friday"
        case 6: res = "Saturday"
        case 7: res = "Sunday"
        case _: res = "Invalid"
    print(f"Result: {res}\n")

def q42(a, b, op):
    print(f"Q42. Operation: {a} {op} {b}")
    match op:
        case '+': res = a + b
        case '-': res = a - b
        case '*': res = a * b
        case '/': res = a / b if b != 0 else "Error: Division by zero"
        case _: res = "Invalid operator"
    print(f"Result: {res}\n")

def q43(month):
    print(f"Q43. Month: {month}")
    match month:
        case "January" | "March" | "May" | "July" | "August" | "October" | "December": res = "31 days"
        case "April" | "June" | "September" | "November": res = "30 days"
        case "February": res = "28 or 29 days"
        case _: res = "Invalid month"
    print(f"Result: {res}\n")
